   374→/* ─── Breadcrumb ─── */
   375→function Breadcrumb({
   376→  segments,
   377→  onNavigate,
   378→}: {
   379→  segments: { label: string; icon?: string }[];
   380→  onNavigate: (index: number) => void;
   381→}) {
   382→  return (
   383→    <nav className="flex items-center gap-1.5 mb-6 overflow-x-auto scrollbar-hide py-1">
   384→      {segments.map((seg, i) => {
   385→        const isLast = i === segments.length - 1;
   386→        return (
   387→          <div key={i} className="flex items-center gap-1.5 flex-shrink-0">
   388→            {i > 0 && (
   389→              <svg
   390→                className="h-3.5 w-3.5 text-zinc-300 dark:text-zinc-600"
   391→                fill="none"
   392→                viewBox="0 0 24 24"
   393→                stroke="currentColor"
   394→                strokeWidth={2}
   395→              >
   396→                <path
   397→                  strokeLinecap="round"
   398→                  strokeLinejoin="round"
   399→                  d="M9 5l7 7-7 7"
   400→                />
   401→              </svg>
   402→            )}
   403→            <button
   404→              onClick={() => !isLast && onNavigate(i)}
   405→              className={`flex items-center gap-1.5 text-sm font-medium transition-colors whitespace-nowrap ${
   406→                isLast
   407→                  ? "text-zinc-900 dark:text-zinc-100 cursor-default"
   408→                  : "text-zinc-400 dark:text-zinc-500 hover:text-zinc-700 dark:hover:text-zinc-300"
   409→              }`}
   410→            >
   411→              {seg.icon && <span>{seg.icon}</span>}
   412→              {seg.label}
   413→            </button>
   414→          </div>
   415→        );
   416→      })}
   417→    </nav>
   418→  );
   419→}

   421→/* ═══════════════════════════════════════════════════════════
   422→   MAIN PAGE
   423→   ═══════════════════════════════════════════════════════════ */
   424→export default function AppStorePage() {
   425→  const [view, setView] = useState<ViewMode>({ kind: "home" });
   426→  const [searchQuery, setSearchQuery] = useState("");
   427→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   428→
   429→  const allApps = categorySections.flatMap((s) => s.apps);
   430→
   431→  const handleNavClick = useCallback((name: string) => {
   432→    setSearchQuery("");
   433→    setMobileMenuOpen(false);
   434→    if (name === "Home") {
   435→      setView({ kind: "home" });
   436→    } else {
   437→      const cat = marketplaceData.find(
   438→        (c) => c.name === name || c.id === name.toLowerCase()
   439→      );
   440→      if (cat) setView({ kind: "topCategory", cat });
   441→    }
   442→  }, []);
   443→
   444→  const handleSearch = useCallback(
   445→    (q: string) => {
   446→      setSearchQuery(q);
   447→      if (q.trim()) {
   448→        setView({ kind: "search", query: q.trim() });
   449→      } else {
   450→        setView({ kind: "home" });
   451→      }
   452→    },
   453→    []
   454→  );
   455→
   456→  // keyboard shortcut: Escape to go back
   457→  useEffect(() => {
   458→    const handler = (e: KeyboardEvent) => {
   459→      if (e.key === "Escape") {
   460→        if (view.kind === "categoryGroup") {
   461→          setView({ kind: "topCategory", cat: view.cat });
   462→        } else if (view.kind === "topCategory" || view.kind === "search") {
   463→          setView({ kind: "home" });
   464→          setSearchQuery("");
   465→        }
   466→      }
   467→    };
   468→    window.addEventListener("keydown", handler);
   469→    return () => window.removeEventListener("keydown", handler);
   470→  }, [view]);
   471→
   472→  // scroll to top on view change
   473→  useEffect(() => {
   474→    window.scrollTo({ top: 0, behavior: "smooth" });
   475→  }, [view.kind]);
   476→
   477→  /* ─── Determine breadcrumbs ─── */
   478→  const breadcrumbs: { label: string; icon?: string }[] = [{ label: "Home", icon: "🏠" }];
   479→  if (view.kind === "topCategory") {
   480→    breadcrumbs.push({ label: view.cat.name, icon: view.cat.icon });
   481→  } else if (view.kind === "categoryGroup") {
   482→    breadcrumbs.push(
   483→      { label: view.cat.name, icon: view.cat.icon },
   484→      { label: view.group.name, icon: view.group.icon }
   485→    );
   486→  } else if (view.kind === "search") {
   487→    breadcrumbs.push({ label: `Search: "${view.query}"`, icon: "🔍" });
   488→  }
   489→
   490→  /* ─── Determine active nav tab ─── */
   491→  const activeTab =
   492→    view.kind === "home"
   493→      ? "Home"
   494→      : view.kind === "topCategory"
   495→        ? view.cat.name
   496→        : view.kind === "categoryGroup"
   497→          ? view.cat.name
   498→          : "Home";
   499→
   500→  /* ─── Search filtered results ─── */
   501→  const filteredApps =
   502→    view.kind === "search"
   503→      ? allApps.filter(
